cmake_minimum_required(VERSION 3.16)
project(xt_stcar_race)
find_package(ament_cmake REQUIRED)
set(STCAR_RACE_BINARY "" CACHE FILEPATH "Rust executable built with --features ros2 for this machine")
if(NOT EXISTS "${STCAR_RACE_BINARY}")
  message(FATAL_ERROR "Set STCAR_RACE_BINARY to the ROS2-enabled Rust executable. See docs/ros2-rust-migration-20261112.md.")
endif()
install(PROGRAMS "${STCAR_RACE_BINARY}" DESTINATION lib/${PROJECT_NAME} RENAME route-engine)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
